cmake_minimum_required(VERSION 3.22)
project(MinecraftOld3dsPatcher VERSION 0.4.11 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(patcherCore src/Bytes.cpp src/Platform.cpp src/Signature.cpp src/Patches.cpp src/UpdatePatches.cpp src/ContainerCancel.cpp)
target_include_directories(patcherCore PUBLIC src)
find_package(Threads REQUIRED)
target_link_libraries(patcherCore PUBLIC Threads::Threads)
if(WIN32)
    target_link_libraries(patcherCore PUBLIC bcrypt)
    target_compile_definitions(patcherCore PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    target_link_libraries(patcherCore PUBLIC OpenSSL::Crypto)
endif()
if(MSVC)
    target_compile_options(patcherCore PRIVATE /W4 /permissive-)
else()
    target_compile_options(patcherCore PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

add_executable(mc3ds-patcher src/main.cpp src/Pipeline.cpp)
target_link_libraries(mc3ds-patcher PRIVATE patcherCore)
target_compile_definitions(mc3ds-patcher PRIVATE MC3DS_VERSION="${PROJECT_VERSION}")
if(MINGW)
    target_link_options(mc3ds-patcher PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()

install(TARGETS mc3ds-patcher RUNTIME DESTINATION .)
install(FILES README.md INPUTS_EXPLAINED.md COMPATIBILITY.md PATCHES.md PERFORMANCE.md DEBUG_OVERLAY.md THIRD_PARTY.md LICENSE DESTINATION .)

include(CTest)
if(BUILD_TESTING)
    add_executable(overlay-tests tests/OverlayTests.cpp patches/OverlayMetrics.cpp)
    target_include_directories(overlay-tests PRIVATE patches)
    add_test(NAME overlay-timing-and-formatting COMMAND overlay-tests)
    add_executable(performance-tests tests/PerformanceTests.cpp)
    target_link_libraries(performance-tests PRIVATE patcherCore)
    add_test(NAME performance-payloads COMMAND performance-tests)
    add_executable(container-cancel-tests tests/ContainerCancelTests.cpp)
    target_link_libraries(container-cancel-tests PRIVATE patcherCore)
    set(MC3DS_UPDATE_FIXTURE "" CACHE PATH "Private stock update extraction for integration tests")
    if(MC3DS_UPDATE_FIXTURE)
        add_test(NAME performance-update-fixture COMMAND performance-tests "${MC3DS_UPDATE_FIXTURE}")
        add_test(NAME container-update-fixture COMMAND container-cancel-tests update "${MC3DS_UPDATE_FIXTURE}")
    endif()
    set(MC3DS_BASE_FIXTURE "" CACHE PATH "Private stock base extraction for integration tests")
    if(MC3DS_BASE_FIXTURE)
        add_test(NAME container-base-fixture COMMAND container-cancel-tests base "${MC3DS_BASE_FIXTURE}")
    endif()
endif()

option(MC3DS_FETCH_TOOLS "Download the pinned upstream CIA utility beside the patcher" OFF)
if(MC3DS_FETCH_TOOLS)
    if(WIN32)
        set(MC3DS_TOOL_PLATFORM windows)
    else()
        set(MC3DS_TOOL_PLATFORM linux)
    endif()
    add_custom_command(TARGET mc3ds-patcher POST_BUILD
        COMMAND ${CMAKE_COMMAND} -DTOOL_PLATFORM=${MC3DS_TOOL_PLATFORM}
        -DDESTINATION=$<TARGET_FILE_DIR:mc3ds-patcher>/tools
        -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/FetchTools.cmake VERBATIM)
endif()
